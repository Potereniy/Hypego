 const b=await bodyJson(req),token=String(b.token||'');
 if(!token)throw new HttpError(400,'Ссылка подтверждения недействительна');
 const rows=await db`SELECT id,user_id FROM email_auth_tokens WHERE purpose='verify_email' AND token_hash=${hashToken(token)} AND consumed_at IS NULL AND expires_at>NOW() LIMIT 1`;
 if(!rows.length)throw new HttpError(400,'Ссылка устарела или уже использована');
 await db`UPDATE users SET email_verified_at=COALESCE(email_verified_at,NOW()) WHERE id=${rows[0].user_id}`;
 await db`UPDATE email_auth_tokens SET consumed_at=NOW() WHERE id=${rows[0].id}`;
 await audit(rows[0].user_id,'auth.email_verified','user',rows[0].user_id);
 return Response.json({ok:true,message:'Email подтверждён. Теперь можно войти.'});
}

if(key==='auth/reset-password'&&method==='POST'){
 const b=await bodyJson(req),token=String(b.token||''),password=String(b.password||''),confirmPassword=String(b.confirmPassword||'');
 if(password.length<10)throw new HttpError(400,'Пароль должен содержать минимум 10 символов');
 if(password!==confirmPassword)throw new HttpError(400,'Пароли не совпадают');
 const rows=await db`SELECT id,user_id FROM email_auth_tokens WHERE purpose='password_reset' AND token_hash=${hashToken(token)} AND consumed_at IS NULL AND expires_at>NOW() LIMIT 1`;
 if(!rows.length)throw new HttpError(400,'Ссылка устарела или уже использована. Запросите новую.');
 const hash=await bcrypt.hash(password,12),uid=rows[0].user_id;
 await db`UPDATE users SET password_hash=${hash},session_version=session_version+1,updated_at=NOW() WHERE id=${uid}`;
 await db`UPDATE email_auth_tokens SET consumed_at=NOW() WHERE user_id=${uid} AND purpose='password_reset' AND consumed_at IS NULL`;
 await db`UPDATE trusted_devices SET revoked_at=NOW() WHERE user_id=${uid} AND revoked_at IS NULL`;
 await clearSession();await clearTrustedDeviceCookie();
 await audit(uid,'auth.password_reset','user',uid);
 return Response.json({ok:true,message:'Пароль изменён. Теперь войдите с новым паролем.'});
}
if(key==='auth/verify-login'&&method==='POST'){
 const b=await bodyJson(req),challengeId=String(b.challengeId||''),code=String(b.code||'').trim();
 if(!/^\d{6}$/.test(code))throw new HttpError(400,'Введите шестизначный код');
 const rows=await db`SELECT id,user_id,token_hash,attempts FROM email_auth_tokens WHERE id=${challengeId} AND purpose='login_code' AND consumed_at IS NULL AND expires_at>NOW() AND attempts<5 LIMIT 1`;
 if(!rows.length)throw new HttpError(400,'Код устарел. Войдите заново и запросите новый.');
 if(rows[0].token_hash!==hashCode(code)){
  await db`UPDATE email_auth_tokens SET attempts=attempts+1 WHERE id=${rows[0].id}`;
  throw new HttpError(400,'Неверный код');
 }
 await db`UPDATE email_auth_tokens SET consumed_at=NOW() WHERE id=${rows[0].id}`;
 const users=await db`SELECT id,username,email,session_version,is_banned FROM users WHERE id=${rows[0].user_id} LIMIT 1`;
 if(!users.length||users[0].is_banned)throw new HttpError(403,'Аккаунт недоступен');
 const chsp=await db`SELECT 1 FROM user_roles WHERE user_id=${users[0].id} AND role_id='chsp' LIMIT 1`;
 if(chsp.length)throw new HttpError(403,'Доступ закрыт: роль ЧСП');
 await trustDevice(String(users[0].id),req.headers.get('user-agent')||'');
 await setSession(await makeSession(users[0]));
 await audit(users[0].id,'auth.login_verified','user',users[0].id);
 return Response.json({user:{id:users[0].id,username:users[0].username,email:users[0].email}});
}
if(key==='auth/register'&&method==='POST'){
 const b=await bodyJson(req),username=String(b.username||'').trim(),email=String(b.email||'').trim().toLowerCase(),password=String(b.password||'');
 if(!/^[A-Za-z0-9_]{3,24}$/.test(username))throw new HttpError(400,'Ник: 3–24 символа, латиница, цифры и _');
 if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email))throw new HttpError(400,'Укажи корректный email');
 if(password.length<10)throw new HttpError(400,'Пароль должен содержать минимум 10 символов');
 const hash=await bcrypt.hash(password,12);
 let rows;
 try{rows=await db`INSERT INTO users(username,email,password_hash,display_name) VALUES(${username},${email},${hash},${username}) RETURNING id,username,email`;}
 catch(e){if((e as any)?.code==='23505')throw new HttpError(409,'Этот никнейм или email уже зарегистрирован');throw e;}
 await db`INSERT INTO user_roles(user_id,role_id) VALUES(${rows[0].id},'player')`;
 const token=randomToken();
 await createEmailToken(String(rows[0].id),'verify_email',token,60);
 const link=appUrl('/verify-email')+'?token='+encodeURIComponent(token);
 try{
  await sendEmail(email,'Подтверждение email — Hypego',`Здравствуйте, ${username}!\nПодтвердите email по ссылке: ${link}\nСсылка действует 60 минут.`,`<p>Здравствуйте, ${username}!</p><p>Подтвердите email для аккаунта Hypego:</p><p><a href="${link}">Подтвердить email</a></p><p>Ссылка действует 60 минут.</p>`);
 }catch(e){console.error('Verification email failed');throw new HttpError(503,'Не удалось отправить письмо. Проверьте настройки SMTP и попробуйте позже.');}
 await audit(rows[0].id,'auth.register','user',rows[0].id);
 return Response.json({verificationRequired:true,message:'Аккаунт создан. Проверьте почту и подтвердите email.'},{status:201});
}
if(key==='auth/login'&&method==='POST'){
 const b=await bodyJson(req),login=String(b.login||'').trim(),password=String(b.password||'');
 const rows=await db`SELECT id,username,email,password_hash,is_banned,email_verified_at,session_version FROM users WHERE lower(email)=lower(${login}) OR lower(username)=lower(${login}) LIMIT 1`;
 if(!rows.length||!await bcrypt.compare(password,rows[0].password_hash))throw new HttpError(401,'Неверный логин или пароль');
 if(rows[0].is_banned)throw new HttpError(403,'Аккаунт заблокирован');
 if(!rows[0].email_verified_at)throw new HttpError(403,'Сначала подтвердите email. Проверьте почту.');
 const chsp=await db`SELECT 1 FROM user_roles WHERE user_id=${rows[0].id} AND role_id='chsp' LIMIT 1`;
 if(chsp.length)throw new HttpError(403,'Доступ закрыт: роль ЧСП');
 const cookieStore=await cookies(),device=cookieStore.get('hypego_device')?.value;
 if(await isTrustedDevice(String(rows[0].id),device,req.headers.get('user-agent')||'')){
  await setSession(await makeSession(rows[0]));
  await audit(rows[0].id,'auth.login','user',rows[0].id);
  return Response.json({user:{id:rows[0].id,username:rows[0].username,email:rows[0].email}});
 }
 const code=String(randomInt(100000,1000000));
 const challengeId=await createLoginCode(String(rows[0].id),code);
 try{await sendEmail(rows[0].email,'Код входа — Hypego',`Ваш код входа: ${code}\nОн действует 10 минут. Если это были не вы, смените пароль.`,`<p>Ваш код входа в Hypego:</p><h2>${code}</h2><p>Код действует 10 минут.</p><p>Если это были не вы, смените пароль.</p>`);}
 catch(e){console.error('Login code email failed');throw new HttpError(503,'Не удалось отправить код. Проверьте настройки почты.');}
 return Response.json({requiresCode:true,challengeId,message:'Код отправлен на вашу почту.'});
